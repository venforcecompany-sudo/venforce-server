-- ============================================================================
-- Verificação pós-incidente P0 (02/10/2026) — SOMENTE LEITURA
--
-- Como rodar: no psql ou no console SQL do banco, o arquivo inteiro.
-- A transação é READ ONLY: qualquer escrita falha com "cannot execute ... in a
-- read-only transaction". Termina em ROLLBACK, não deixa nada para trás.
--
-- Janela: 29/09/2026 (deploy do PR #195) em diante. Ajuste o literal abaixo
-- (procure por JANELA) se quiser olhar menos.
-- Fuso de negócio: America/Sao_Paulo. Colunas TIMESTAMPTZ aparecem em BRT.
-- ============================================================================
BEGIN READ ONLY;
SET LOCAL statement_timeout = '30s';                    -- teto por comando
SET LOCAL lock_timeout = '3s';                          -- nunca espera lock de DDL/escrita alheia
SET LOCAL idle_in_transaction_session_timeout = '120s'; -- se o terminal travar, o banco derruba a sessão
SET LOCAL TIME ZONE 'America/Sao_Paulo';

-- ---------------------------------------------------------------------------
-- Q1 — Runs AINDA ativos (queued/running) agora. Separa manual × noturno.
-- manual = requested_by preenchido (clique em "Atualizar dados" ou Central);
-- noturno = requested_by NULL (scheduler).
-- idade > 60 min em running ou > 15 min em queued = órfão (o próximo clique
-- do mesmo escopo o fecha sozinho como SYNC_RUN_STALE_*).
-- ---------------------------------------------------------------------------
SELECT r.id, r.cliente_slug, r.cliente_conta_id, c.nome AS conta,
       r.status,
       CASE WHEN r.requested_by IS NULL THEN 'noturno' ELSE 'manual' END AS origem,
       r.requested_by, r.date_from, r.date_to,
       r.created_at, r.started_at,
       NOW() - COALESCE(r.started_at, r.created_at) AS idade,
       CASE
         WHEN r.status = 'queued'  AND r.created_at < NOW() - INTERVAL '15 minutes' THEN 'ORFAO_QUEUED'
         WHEN r.status = 'running' AND r.started_at < NOW() - INTERVAL '60 minutes' THEN 'ORFAO_RUNNING'
         ELSE 'dentro_do_prazo'
       END AS classificacao
  FROM central_vendas_sync_runs r
  LEFT JOIN cliente_contas c ON c.id = r.cliente_conta_id
 WHERE r.status IN ('queued', 'running')
 ORDER BY r.created_at;

-- ---------------------------------------------------------------------------
-- Q2 — Panorama da janela: status e código de erro por origem e dia (BRT).
-- Mostra quando os runs pararam de terminar (incidente) e o que cada
-- mecanismo de limpeza fechou (SYNC_RUN_STALE_*, SYNC_RUN_PROCESS_RESTART).
-- ---------------------------------------------------------------------------
SELECT (r.created_at AT TIME ZONE 'America/Sao_Paulo')::date AS dia,
       CASE WHEN r.requested_by IS NULL THEN 'noturno' ELSE 'manual' END AS origem,
       r.status, r.error_code,
       COUNT(*) AS runs,
       MIN(r.created_at) AS primeiro, MAX(r.created_at) AS ultimo
  FROM central_vendas_sync_runs r
 WHERE r.created_at >= TIMESTAMPTZ '2026-09-29 00:00:00-03'   -- JANELA
 GROUP BY 1, 2, 3, 4
 ORDER BY 1, 2, 3, 4;

-- ---------------------------------------------------------------------------
-- Q3 — Runs encerrados à força pelo sistema (não por terminarem de fato).
-- Esses runs NÃO publicam. Lista o que foi interrompido e a duração real.
-- ---------------------------------------------------------------------------
SELECT r.id, r.cliente_slug, r.cliente_conta_id, r.status, r.error_code,
       CASE WHEN r.requested_by IS NULL THEN 'noturno' ELSE 'manual' END AS origem,
       r.date_from, r.date_to, r.created_at, r.started_at, r.finished_at,
       r.finished_at - COALESCE(r.started_at, r.created_at) AS duracao
  FROM central_vendas_sync_runs r
 WHERE r.error_code IN ('SYNC_RUN_STALE_QUEUED', 'SYNC_RUN_STALE_RUNNING', 'SYNC_RUN_PROCESS_RESTART')
   AND r.created_at >= TIMESTAMPTZ '2026-09-29 00:00:00-03'   -- JANELA
 ORDER BY r.created_at;

-- ---------------------------------------------------------------------------
-- Q4 — Runs MANUAIS da janela: desfecho, completude e o que produziram.
-- candidatos = imports 'candidate' ainda não promovidos; publicados = imports
-- 'published' deste run. Run completed COM candidatos e SEM publicados é o
-- sinal de "terminou mas não promoveu" (ver Q6).
-- ---------------------------------------------------------------------------
SELECT r.id, r.cliente_slug, r.cliente_conta_id, r.status, r.completeness_status,
       r.error_code, r.date_from, r.date_to, r.created_at, r.finished_at,
       COUNT(i.id) FILTER (WHERE i.publication_status = 'candidate') AS imports_candidate,
       COUNT(i.id) FILTER (WHERE i.publication_status = 'published') AS imports_published
  FROM central_vendas_sync_runs r
  LEFT JOIN central_vendas_imports i ON i.sync_run_id = r.id
 WHERE r.requested_by IS NOT NULL
   AND r.created_at >= TIMESTAMPTZ '2026-09-29 00:00:00-03'   -- JANELA
 GROUP BY r.id
 ORDER BY r.created_at;

-- ---------------------------------------------------------------------------
-- Q5 — Fontes (orders/shipments/...) de runs manuais que NÃO fecharam.
-- Mostra até onde cada run interrompido chegou.
-- ---------------------------------------------------------------------------
SELECT r.id AS run_id, r.status AS run_status, s.source, s.status AS source_status,
       s.complete, s.expected_count, s.received_count, s.error_code, s.finished_at
  FROM central_vendas_sync_runs r
  JOIN central_vendas_sync_sources s ON s.sync_run_id = r.id
 WHERE r.requested_by IS NOT NULL
   AND r.status <> 'completed'
   AND r.created_at >= TIMESTAMPTZ '2026-09-29 00:00:00-03'   -- JANELA
 ORDER BY r.id, s.source;

-- ---------------------------------------------------------------------------
-- Q6 — INCONSISTÊNCIAS DE PUBLICAÇÃO (o que não deveria existir).
-- (a) import 'published' cujo run não está 'completed'  → deveria ser 0 linhas
-- (b) import 'candidate' de run 'completed' (nunca promovido) → depende: run
--     completed mas parcial/incompleto não promove de propósito; investigue
-- (c) import 'candidate' de run 'failed'/'queued'/'running' → esperado (lixo
--     inerte: nunca é lido), só dimensiona
-- ---------------------------------------------------------------------------
SELECT 'a_published_com_run_nao_completed' AS tipo, i.id AS import_id, i.sync_run_id,
       r.status AS run_status, i.cliente_slug, i.cliente_conta_id, i.competencia,
       i.publication_status, i.published_at
  FROM central_vendas_imports i
  JOIN central_vendas_sync_runs r ON r.id = i.sync_run_id
 WHERE i.publication_status = 'published' AND r.status <> 'completed'
UNION ALL
SELECT 'b_candidate_de_run_completed', i.id, i.sync_run_id, r.status, i.cliente_slug,
       i.cliente_conta_id, i.competencia, i.publication_status, i.published_at
  FROM central_vendas_imports i
  JOIN central_vendas_sync_runs r ON r.id = i.sync_run_id
 WHERE i.publication_status = 'candidate' AND r.status = 'completed'
   AND r.created_at >= TIMESTAMPTZ '2026-09-29 00:00:00-03'   -- JANELA
ORDER BY 1, 2;

SELECT r.status AS run_status, i.publication_status, COUNT(*) AS imports
  FROM central_vendas_imports i
  JOIN central_vendas_sync_runs r ON r.id = i.sync_run_id
 WHERE r.created_at >= TIMESTAMPTZ '2026-09-29 00:00:00-03'   -- JANELA
 GROUP BY 1, 2
 ORDER BY 1, 2;

-- ---------------------------------------------------------------------------
-- Q7 — "Duplicação" nos dados publicados.
-- O desenho PERMITE vários imports published para a mesma conta×competência
-- (cada run completo publica o seu; a leitura escolhe UM, o de maior
-- published_at). Então >1 published NÃO é erro por si só. O que é anomalia:
--   (a) dois published com o MESMO published_at (leitura ambígua);
--   (b) pedido repetido DENTRO de um mesmo import (UNIQUE impede; não consultado);
--   (c) published sem nenhum pedido.
-- ---------------------------------------------------------------------------
-- (a) empate de published_at na mesma conta×competência
SELECT i.cliente_slug, i.cliente_conta_id, i.competencia, i.published_at,
       COUNT(*) AS imports_empatados, ARRAY_AGG(i.id ORDER BY i.id) AS import_ids
  FROM central_vendas_imports i
 WHERE i.publication_status = 'published'
 GROUP BY 1, 2, 3, 4
HAVING COUNT(*) > 1
 ORDER BY 3 DESC, 1;

-- (b) pedido repetido dentro do mesmo import: IMPOSSÍVEL (UNIQUE (import_id,
--     pedido_id) no schema); consulta removida por ser pesada e inútil.

-- (c) published sem pedido nenhum, na janela
SELECT i.id AS import_id, i.sync_run_id, i.cliente_slug, i.cliente_conta_id,
       i.competencia, i.published_at
  FROM central_vendas_imports i
 WHERE i.publication_status = 'published'
   AND i.created_at >= TIMESTAMPTZ '2026-09-29 00:00:00-03'   -- JANELA
   AND NOT EXISTS (SELECT 1 FROM central_vendas_pedidos p WHERE p.import_id = i.id)
 ORDER BY i.id;

-- ---------------------------------------------------------------------------
-- Q8 — Pagamentos do Mercado Pago: gravados por sync_run_id (UNIQUE
-- (sync_run_id, payment_id) impede duplicata dentro do run). Dimensiona o
-- que ficou em runs que NÃO completaram (inerte, nunca publicado).
-- ---------------------------------------------------------------------------
SELECT r.status AS run_status, COUNT(DISTINCT r.id) AS runs, COUNT(p.id) AS payments
  FROM central_vendas_sync_runs r
  JOIN central_vendas_mp_payments p ON p.sync_run_id = r.id
 WHERE r.created_at >= TIMESTAMPTZ '2026-09-29 00:00:00-03'   -- JANELA
 GROUP BY 1
 ORDER BY 1;

SELECT r.status AS run_status, rep.status AS repasse_status, COUNT(*) AS relatorios
  FROM central_vendas_sync_runs r
  JOIN central_vendas_mp_settlement_reports rep ON rep.sync_run_id = r.id
 WHERE r.created_at >= TIMESTAMPTZ '2026-09-29 00:00:00-03'   -- JANELA
 GROUP BY 1, 2
 ORDER BY 1, 2;

-- ---------------------------------------------------------------------------
-- Q9 — Resumos derivados (upsert): Cliente 360 e Ads.
-- UNIQUE (cliente_id, competencia) e (cliente_slug, mes_ref, loja_campanha)
-- impedem duplicata. O que checar é DEFASAGEM: resumo C360 mais velho que o
-- último import published da mesma competência (é o critério que o scheduler
-- usa para reprocessar).
-- ---------------------------------------------------------------------------
SELECT s.cliente_slug, s.competencia, s.sincronizado_em AS resumo_sincronizado_em,
       MAX(i.published_at) AS ultimo_published,
       MAX(i.published_at) - s.sincronizado_em AS defasagem
  FROM cliente_360_resumos_mensais s
  JOIN central_vendas_imports i
    ON i.cliente_slug = s.cliente_slug
   AND i.competencia = s.competencia
   AND i.publication_status = 'published'
 GROUP BY s.cliente_slug, s.competencia, s.sincronizado_em
HAVING MAX(i.published_at) > COALESCE(s.sincronizado_em, 'epoch'::timestamptz)
 ORDER BY defasagem DESC
 LIMIT 100;

-- Ads atualizado na janela (só para confirmar que o upsert não multiplicou linhas)
SELECT cliente_slug, mes_ref, COUNT(*) AS linhas, MAX(updated_at) AS ultima_atualizacao
  FROM ads_resumos_mensais
 WHERE updated_at >= TIMESTAMPTZ '2026-09-29 00:00:00-03'   -- JANELA
 GROUP BY 1, 2
HAVING COUNT(*) > 1
 ORDER BY 1, 2;

-- ---------------------------------------------------------------------------
-- Q10 — Agendador noturno: o que a rodada de 02/10 03:00 (BRT) deveria ter
-- feito versus o que existe. Dia 2 => períodos: setembro/2026 completo e
-- outubro/2026 01..01 (mês corrente até ontem). Ajuste as datas em outros dias.
-- Contas MELI ativas SEM run noturno criado desde 02/10 03:00. Aproximação:
-- inclui contas que o scheduler considera inelegíveis (sem grant etc.).
-- ---------------------------------------------------------------------------
SELECT c.id AS conta_id, c.cliente_id, c.nome,
       p.date_from, p.date_to,
       EXISTS (
         SELECT 1 FROM central_vendas_sync_runs r
          WHERE r.cliente_conta_id = c.id AND r.requested_by IS NULL
            AND r.date_from = p.date_from AND r.date_to = p.date_to
            AND r.created_at >= TIMESTAMPTZ '2026-10-02 03:00:00-03'
       ) AS tem_run_noturno
  FROM cliente_contas c
  JOIN clientes cl ON cl.id = c.cliente_id AND cl.ativo = true
 CROSS JOIN (VALUES (DATE '2026-09-01', DATE '2026-09-30'),
                    (DATE '2026-10-01', DATE '2026-10-01')) AS p(date_from, date_to)
 WHERE c.marketplace = 'meli' AND c.ativo = true
 ORDER BY tem_run_noturno, c.id, p.date_from;

-- Resumo dos runs noturnos por dia/status (últimos 7 dias)
SELECT (r.created_at AT TIME ZONE 'America/Sao_Paulo')::date AS dia,
       r.status, r.error_code, COUNT(*) AS runs
  FROM central_vendas_sync_runs r
 WHERE r.requested_by IS NULL
   AND r.created_at >= NOW() - INTERVAL '7 days'
 GROUP BY 1, 2, 3
 ORDER BY 1, 2, 3;

-- ---------------------------------------------------------------------------
-- Q11 — O QUE O PRÓXIMO BOOT TENTARIA RECUPERAR (mesma query de
-- listarPeriodosNoturnosPendentes, com antesDe = agora). Se retornar linhas,
-- um novo restart dispararia a recuperação — que hoje falha por causa do bug
-- do tipo DATE (ver relatório). Esperado após a falha de hoje: poucas/nenhuma.
-- ---------------------------------------------------------------------------
SELECT DISTINCT r.date_from, r.date_to
  FROM central_vendas_sync_runs r
 WHERE r.requested_by IS NULL
   AND r.created_at < NOW()
   AND (
     r.status IN ('queued', 'running')
     OR (
       r.status = 'completed'
       AND EXISTS (
         SELECT 1 FROM central_vendas_imports i
          WHERE i.sync_run_id = r.id AND i.publication_status = 'published'
       )
       AND NOT EXISTS (
         SELECT 1 FROM cliente_360_resumos_mensais s
          WHERE s.cliente_id = r.cliente_id
            AND s.competencia = TO_CHAR(r.date_from, 'YYYY-MM')
            AND s.sincronizado_em >= COALESCE((
              SELECT MAX(i2.published_at) FROM central_vendas_imports i2
               WHERE i2.sync_run_id = r.id AND i2.publication_status = 'published'
            ), r.finished_at)
       )
     )
   )
 ORDER BY r.date_from, r.date_to;

-- ---------------------------------------------------------------------------
-- Q12 — Sessões do banco agora (confirma que o pool não está preso e que as
-- conexões de lock do Painel só existem enquanto há atualização em curso).
-- ---------------------------------------------------------------------------
SELECT application_name, state, wait_event_type, wait_event, COUNT(*) AS sessoes,
       MAX(NOW() - state_change) AS mais_antiga
  FROM pg_stat_activity
 WHERE datname = current_database()
 GROUP BY 1, 2, 3, 4
 ORDER BY sessoes DESC;

ROLLBACK;

-- ============================================================================
-- Verificação pós-incidente P0 — PARTE 2 — SOMENTE LEITURA
-- Mesmas travas da parte 1: BEGIN READ ONLY, statement_timeout 30s,
-- lock_timeout 3s, idle_in_transaction 120s, ROLLBACK no final.
-- Os ids dos runs vêm do relatório da parte 1 (verificacao_pos_incidente_saida.txt).
-- ============================================================================
BEGIN READ ONLY;
SET LOCAL statement_timeout = '30s';
SET LOCAL lock_timeout = '3s';
SET LOCAL idle_in_transaction_session_timeout = '120s';
SET LOCAL TIME ZONE 'America/Sao_Paulo';

-- ---------------------------------------------------------------------------
-- R1 — LACUNA REAL DE SETEMBRO: contas MELI ativas SEM import published cuja
-- cobertura contenha 2026-09-01..2026-09-30 inteiro. A leitura só aceita um
-- published que contenha o trecho pedido inteiro; sem ele cai no legado/vazio.
-- ultima_cobertura_ate mostra até onde o melhor published chega.

  SELECT COUNT(*) INTO n_ok FROM alvo_orfaos a JOIN central_vendas_sync_runs r ON r.id = a.id
   WHERE r.status = a.status_esperado
       AND (a.origem = 'noturno') = (r.requested_by IS NULL)
       AND r.created_at < TIMESTAMPTZ '2026-10-02 01:00:00-03'
       AND (   (r.status = 'queued'  AND r.started_at IS NULL AND r.created_at < NOW() - INTERVAL '15 minutes')
            OR (r.status = 'running' AND r.started_at < NOW() - INTERVAL '60 minutes'));
  IF n_ok <> n_alvo THEN
    RAISE EXCEPTION 'G1: so % de % alvos estao no estado esperado (status/origem/idade/corte); abortando', n_ok, n_alvo;
  END IF;

-- e repete o predicado de G1 (status/idade) no próprio UPDATE — fecha a janela de corrida.
WITH travados AS (
  SELECT r.id FROM central_vendas_sync_runs r JOIN alvo_orfaos a ON a.id = r.id
   WHERE r.status = a.status_esperado
       AND (a.origem = 'noturno') = (r.requested_by IS NULL)
       AND r.created_at < TIMESTAMPTZ '2026-10-02 01:00:00-03'
       AND (   (r.status = 'queued'  AND r.started_at IS NULL AND r.created_at < NOW() - INTERVAL '15 minutes')
            OR (r.status = 'running' AND r.started_at < NOW() - INTERVAL '60 minutes'))
   ORDER BY r.id
     FOR UPDATE OF r NOWAIT
), alterados AS (
  UPDATE central_vendas_sync_runs r
     SET status = 'failed', finished_at = NOW(), updated_at = NOW(),
         error_code = 'SYNC_RUN_ORPHAN_POST_INCIDENT_20261002',
         error_message = 'Run orfao encerrado manualmente apos o incidente P0 de 2026-10-02: processo morto, sem worker ativo. Nenhum dado foi publicado por este run.'
    FROM travados t
   WHERE r.id = t.id
  RETURNING r.id, r.cliente_slug, r.cliente_conta_id, r.date_from, r.date_to, r.requested_by
)
SELECT a.origem, a.status_esperado AS status_anterior, COUNT(*) AS runs_encerrados,
       MIN(al.id) AS menor_id, MAX(al.id) AS maior_id
  FROM alterados al JOIN alvo_orfaos a ON a.id = al.id
 GROUP BY 1, 2 ORDER BY 1, 2;

-- G2 — conferência DEPOIS (ainda dentro da transação).
DO $$
DECLARE n_falhos INT; n_codigo INT; n_ativos_alvo INT; n_alvo INT;
BEGIN

  SELECT COUNT(*) INTO n_falhos FROM alvo_orfaos a JOIN central_vendas_sync_runs r ON r.id = a.id
   WHERE r.status = 'failed' AND r.error_code = 'SYNC_RUN_ORPHAN_POST_INCIDENT_20261002' AND r.finished_at IS NOT NULL;
  SELECT COUNT(*) INTO n_codigo FROM central_vendas_sync_runs WHERE error_code = 'SYNC_RUN_ORPHAN_POST_INCIDENT_20261002';
  SELECT COUNT(*) INTO n_ativos_alvo FROM alvo_orfaos a JOIN central_vendas_sync_runs r ON r.id = a.id
   WHERE r.status IN ('queued','running');
  IF n_falhos <> n_alvo THEN RAISE EXCEPTION 'G2: % de % alvos encerrados', n_falhos, n_alvo; END IF;
  IF n_codigo <> n_alvo THEN RAISE EXCEPTION 'G2: % linhas com o codigo (esperado %): alguma linha fora do alvo foi tocada', n_codigo, n_alvo; END IF;
  IF n_ativos_alvo <> 0 THEN RAISE EXCEPTION 'G2: % alvos ainda ativos', n_ativos_alvo; END IF;
END $$;

-- Panorama final dos runs ativos que SOBRARAM (esperado: 472 e eventuais runs legítimos novos).
SELECT r.id, r.cliente_slug, r.status, CASE WHEN r.requested_by IS NULL THEN 'noturno' ELSE 'manual' END AS origem,
       r.created_at, NOW() - COALESCE(r.started_at, r.created_at) AS idade
  FROM central_vendas_sync_runs r WHERE r.status IN ('queued','running') ORDER BY r.created_at;

-- DRY-RUN: executa o UPDATE de verdade dentro da transação e DESFAZ no fim.
ROLLBACK;

8.1 _conexao.sh (biblioteca de conexão; carregada por source)
--------------------------------------------------------------------------------
Lê a URL em prompt oculto; modo ro exige e confirma default_transaction_read_only=on.

# Biblioteca de conexão (carregada com `source`). Lê a DATABASE_URL num prompt oculto, decompõe em variáveis
# PG* (só neste processo) e NUNCA a coloca em linha de comando, arquivo ou histórico.
# uso: conectar ro|rw   — ro: sessão inteira somente-leitura (exigido e confirmado); rw: escrita habilitada.
conectar() {
  local modo="${1:-ro}"
  command -v psql >/dev/null    || { echo "ERRO: psql nao encontrado"; exit 1; }
  command -v python3 >/dev/null || { echo "ERRO: python3 nao encontrado"; exit 1; }
  if [ -z "${DATABASE_URL:-}" ]; then
    if [ -t 0 ]; then read -rsp "Cole a DATABASE_URL de producao (nao aparece na tela) e tecle Enter: " DATABASE_URL; echo
    else read -r DATABASE_URL; fi
  fi
  case "${DATABASE_URL:-}" in postgres://*|postgresql://*) ;; *) echo "ERRO: a URL deve comecar com postgresql://"; exit 1;; esac
  eval "$(DATABASE_URL="$DATABASE_URL" python3 - <<'PY'
import os, shlex
from urllib.parse import urlsplit, unquote, parse_qs
u = urlsplit(os.environ["DATABASE_URL"]); q = parse_qs(u.query)
v = {"PGHOST": u.hostname or "", "PGPORT": str(u.port or 5432), "PGUSER": unquote(u.username or ""),
     "PGPASSWORD": unquote(u.password or ""), "PGDATABASE": unquote((u.path or "/").lstrip("/")),
     "PGSSLMODE": q.get("sslmode", ["require"])[0]}
print("\n".join("export %s=%s" % (k, shlex.quote(x)) for k, x in v.items()))
PY
)"
  unset DATABASE_URL
  [ -n "${PGHOST:-}" ] && [ -n "${PGUSER:-}" ] && [ -n "${PGDATABASE:-}" ] || { echo "ERRO: URL incompleta (host/usuario/banco)"; exit 1; }
  export PGCONNECT_TIMEOUT=15
  if [ "$modo" = "ro" ]; then export PGOPTIONS='-c default_transaction_read_only=on'; else unset PGOPTIONS; fi
  local v; v="$(psql -X -At -v ON_ERROR_STOP=1 -c 'SHOW default_transaction_read_only' 2>&1)" || { echo "ERRO de conexao: $(printf '%s' "$v" | head -2)"; exit 1; }
  if [ "$modo" = "ro" ]; then
    [ "$v" = "on" ] || { echo "ABORTADO: a sessao NAO esta somente-leitura (valor: $v)"; exit 1; }
    echo "Conexao OK (host $PGHOST) — sessao SOMENTE-LEITURA confirmada."
  else
    echo "Conexao OK (host $PGHOST) — modo ESCRITA habilitado: o dry-run desfaz tudo com ROLLBACK."
  fi
}

BEGIN
  SELECT COUNT(*) INTO n_alvo FROM alvo_orfaos;
  IF n_alvo <> 85 THEN RAISE EXCEPTION 'G1: esperado 85 alvos declarados, achei %', n_alvo; END IF;
  SELECT COUNT(*) INTO n_conf FROM alvo_orfaos a JOIN central_vendas_sync_runs r ON r.id = a.id WHERE r.status = a.status_esperado

   WHERE r.status = a.status_esperado

), alterados AS (

         error_message = 'Run orfao encerrado manualmente apos o incidente P0 de 2026-10-02: processo morto, sem worker ativo. Nenhum dado foi publicado por este run.'

)

  SELECT COUNT(*) INTO n_cod FROM central_vendas_sync_runs WHERE error_code = 'SYNC_RUN_ORPHAN_POST_INCIDENT_20261002' AND status = 'failed';

END $$;

--- 9.3 q60_DRYRUN.sql (termina em ROLLBACK; NÃO executado) — o q60_APLICAR.sql é idêntico exceto o rodapé COMMIT; ---
-- ============================================================================
-- ENCERRAMENTO DOS 60 RUNS NOTURNOS QUEUED DE 03/10 — INCIDENTE P0 2026-10-02 (nao executado)
-- Altera APENAS central_vendas_sync_runs (status/finished_at/updated_at/error_*) dos runs queued criados pela
-- rodada de 2026-10-03 03:00 (requested_by NULL, started_at NULL). Nada e deletado; nenhuma outra tabela.
-- Alvo por PREDICADO (os ids ficam no backup), com portoes: exatamente 60, nenhum running, nenhum manual ativo,
-- nenhum advisory lock do agendador, nenhum alvo com menos de 15 min. Qualquer divergencia aborta tudo.
-- Rodar com:  psql -X -v ON_ERROR_STOP=1 -f <arquivo>   (conexao por variaveis PG*)
-- ============================================================================
BEGIN;
SET LOCAL statement_timeout = '30s';
SET LOCAL lock_timeout = '3s';
SET LOCAL idle_in_transaction_session_timeout = '300s';
SET LOCAL TIME ZONE 'America/Sao_Paulo';

CREATE TEMP TABLE alvo_q60 ON COMMIT DROP AS
SELECT id FROM central_vendas_sync_runs
 WHERE status = 'queued' AND requested_by IS NULL AND started_at IS NULL
   AND created_at >= TIMESTAMPTZ '2026-10-03 03:00:00-03' AND created_at < TIMESTAMPTZ '2026-10-03 03:01:00-03';

-- G1 — conferencia ANTES.
DO $$
DECLARE n_alvo INT; n_run INT; n_man INT; n_lock INT; n_novos INT; n_cod INT; n_fora INT;
BEGIN
  SELECT COUNT(*) INTO n_alvo FROM alvo_q60;
  IF n_alvo <> 60 THEN RAISE EXCEPTION 'G1: esperado 60 alvos, achei % (abortando)', n_alvo; END IF;
  SELECT COUNT(*) INTO n_run FROM central_vendas_sync_runs WHERE status = 'running';
  IF n_run <> 0 THEN RAISE EXCEPTION 'G1: ha % runs running agora (abortando)', n_run; END IF;
  SELECT COUNT(*) INTO n_man FROM central_vendas_sync_runs WHERE requested_by IS NOT NULL AND status IN ('queued','running');
  IF n_man <> 0 THEN RAISE EXCEPTION 'G1: ha % runs manuais ativos (abortando)', n_man; END IF;
  SELECT COUNT(*) INTO n_lock FROM pg_locks WHERE locktype = 'advisory' AND classid = 1296845920;
  IF n_lock <> 0 THEN RAISE EXCEPTION 'G1: o lock global do agendador esta em uso (rodada em andamento?) (abortando)'; END IF;
  SELECT COUNT(*) INTO n_novos FROM alvo_q60 a JOIN central_vendas_sync_runs r ON r.id = a.id WHERE r.created_at > NOW() - INTERVAL '15 minutes';
  IF n_novos <> 0 THEN RAISE EXCEPTION 'G1: % alvos tem menos de 15 min (abortando)', n_novos; END IF;
  SELECT COUNT(*) INTO n_fora FROM central_vendas_sync_runs r WHERE r.status = 'queued' AND NOT EXISTS (SELECT 1 FROM alvo_q60 a WHERE a.id = r.id);
  IF n_fora <> 0 THEN RAISE EXCEPTION 'G1: ha % runs queued FORA do alvo (aparecem runs novos; reavaliar) (abortando)', n_fora; END IF;
  SELECT COUNT(*) INTO n_cod FROM central_vendas_sync_runs WHERE error_code = 'SYNC_RUN_ORPHAN_NOTURNO_20261003';
  IF n_cod <> 0 THEN RAISE EXCEPTION 'G1: ja existem % runs com o codigo SYNC_RUN_ORPHAN_NOTURNO_20261003 (ja aplicado?) (abortando)', n_cod; END IF;
END $$;

-- UPDATE: NOWAIT (se algum worker vivo segura a linha, aborta em vez de esperar); predicado repetido.
WITH travados AS (
  SELECT r.id FROM central_vendas_sync_runs r JOIN alvo_q60 a ON a.id = r.id
   WHERE r.status = 'queued' AND r.started_at IS NULL AND r.requested_by IS NULL
   ORDER BY r.id FOR UPDATE OF r NOWAIT
), alterados AS (
  UPDATE central_vendas_sync_runs r
     SET status = 'failed', finished_at = NOW(), updated_at = NOW(),
         error_code = 'SYNC_RUN_ORPHAN_NOTURNO_20261003',
         error_message = 'Run noturno queued abandonado: a rodada de 2026-10-03 ficou presa e foi interrompida por restart; encerrado manualmente para impedir uma retomada em massa. Nenhum dado foi publicado por este run.'
    FROM travados t WHERE r.id = t.id
  RETURNING r.id
)
SELECT COUNT(*) AS runs_encerrados, MIN(id) AS menor_id, MAX(id) AS maior_id FROM alterados;

-- G2 — conferencia DEPOIS (ainda dentro da transacao).
DO $$
DECLARE n_cod INT; n_ativos INT; n_alvo_nao_failed INT;
BEGIN
  SELECT COUNT(*) INTO n_cod FROM central_vendas_sync_runs WHERE error_code = 'SYNC_RUN_ORPHAN_NOTURNO_20261003' AND status = 'failed' AND finished_at IS NOT NULL;
  IF n_cod <> 60 THEN RAISE EXCEPTION 'G2: % linhas com o codigo (esperado 60)', n_cod; END IF;
  SELECT COUNT(*) INTO n_alvo_nao_failed FROM alvo_q60 a JOIN central_vendas_sync_runs r ON r.id = a.id WHERE r.status <> 'failed';
  IF n_alvo_nao_failed <> 0 THEN RAISE EXCEPTION 'G2: % alvos nao estao failed', n_alvo_nao_failed; END IF;
  SELECT COUNT(*) INTO n_ativos FROM central_vendas_sync_runs WHERE status IN ('queued','running');
  IF n_ativos <> 0 THEN RAISE EXCEPTION 'G2: sobraram % runs ativos', n_ativos; END IF;
  RAISE NOTICE 'G2 OK: 60 encerrados, 0 runs ativos restantes';
END $$;

ROLLBACK;

--- 9.5 setembro_plano.sql (SOMENTE LEITURA; NÃO executado) ---
-- PLANO DE SETEMBRO/2026 — SOMENTE LEITURA (SELECT; termina em ROLLBACK). Gera a lista por CLIENTE (o --clientes do
-- backfill filtra por cliente), com volume de referencia e as contas incompletas. Saida contem slugs: NAO publicar.
BEGIN READ ONLY;
SET LOCAL statement_timeout = '30s';
SET LOCAL lock_timeout = '3s';
SET LOCAL TIME ZONE 'America/Sao_Paulo';
\pset format unaligned
\pset fieldsep '|'
\pset footer off
WITH contas AS (
  SELECT cc.id AS conta_id, cc.cliente_id, c.slug,
         EXISTS (SELECT 1 FROM central_vendas_imports i
                  WHERE i.cliente_conta_id = cc.id AND i.competencia = '2026-09' AND i.publication_status = 'published'
                    AND i.coverage_date_from <= DATE '2026-09-01' AND i.coverage_date_to >= DATE '2026-09-30') AS completo,
         COALESCE((SELECT MAX((SELECT COUNT(*) FROM central_vendas_pedidos p WHERE p.import_id = i.id))
                     FROM central_vendas_imports i
                    WHERE i.cliente_conta_id = cc.id AND i.competencia = '2026-09'), 0) AS pedidos_ref
    FROM cliente_contas cc
    JOIN clientes c ON c.id = cc.cliente_id AND c.ativo = true
   WHERE cc.marketplace = 'meli' AND cc.ativo = true AND COALESCE(TRIM(cc.external_account_id), '') <> ''
)

       COUNT(*) AS contas_elegiveis,
       COUNT(*) FILTER (WHERE NOT completo) AS contas_incompletas,
       SUM(pedidos_ref)::bigint AS pedidos_ref,
       BOOL_OR(conta_id = 81) AS tem_conta_81,
       string_agg(conta_id::text, ',' ORDER BY conta_id) FILTER (WHERE NOT completo) AS ids_incompletas
  FROM contas
 GROUP BY slug
HAVING COUNT(*) FILTER (WHERE NOT completo) > 0
 ORDER BY BOOL_OR(conta_id = 81), SUM(pedidos_ref), slug;
ROLLBACK;
